public class Main 
{
    public static void main(String[] args)
    {
        int count=0;
        String a = "ragul";
        for(int i=0;i<a.length();i++)
        {
            for(int j=1;j<a.length()-1;j++)
            {
                if(a.charAt(i)==a.charAt(j))
                {
                    System.out.println(a.charAt(i));
                    return;
                }
                count++;
            }
        }
        if(count>0)
        System.out.println("there is no repeating character");
    }
}
