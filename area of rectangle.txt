import java.util.Scanner;
class rect
{
    int a;
    int c;
    rect(int l,int b)
    {
        a=l;
        c=b;
    }
    public void display()
    {
        System.out.println("area is "+(a*c));
    }
}


public class Main 
{
    public static void main(String[] args)
    {
        int l,b;
        Scanner a= new Scanner(System.in);
        System.out.println("Enter the number of rectangle:");
        int n = a.nextInt();
        rect[] r= new rect[n];      
        for(int i=0;i<n;i++)
        {
            System.out.println("Enter the lenght of the rectangle"+(i+1));
            l=a.nextInt();
            System.out.println("Enter the breadth of the rectangle"+(i+1));
            b=a.nextInt();
            r[i]= new rect(l,b);
        }
        for (int i=0;i<n;i++)
        {
            System.out.println("the area for all the rectangle");
            r[i].display();
        }

    }
}
