import java.util.Scanner;
class percentage
{
    int a;
    int b;
    int c;
    percentage(int m1,int m2,int m3)
    {
        a=m1;
        b=m2;
        c=m3;
    }
    public void display()
    {
        System.out.println(""+((a + b + c) / 300.0) * 100);
    }
    
}
public class Main 
{
    public static void main(String[] args)
    {
        int m1,m2,m3;
        Scanner a= new Scanner(System.in);
        System.out.println("Enter the number of Students:");
        int n = a.nextInt();
        percentage[] r= new percentage[n];      
        for(int i=0;i<n;i++)
        {
            System.out.println("Enter the marks of student "+(i+1));
            
            System.out.println("Enter the marks of subject 1");
            m1 = a.nextInt();
            
            System.out.println("Enter the marks of subject 2");
            m2 = a.nextInt();
            System.out.println("Enter the marks of subject 3");
            m3 = a.nextInt();
           r[i]= new percentage(m1,m2,m3);
        }
        for (int i=0;i<n;i++)
        {
            System.out.println("the percentage of the student "+(i+1)+" in  SATHYABAMA is ");
            r[i].display();
        }

    }
}
