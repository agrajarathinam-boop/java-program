public class Main 
{
    public static int rag(int a)
    {
        int s=0;
        while(a>0)
        {
            s=s+(a%10);
            a=a/10;
        }
        return s;
    }
    public static boolean prime(int b)
    {
        if(b<2)
         return false;
        for(int i=2;i<b;i++)
        {
            if(b%i==0)
            return false;
        }
        return true;
    }
    public static void main(String[] arg)
    {
        int s;
        for (int i=100;i<1000;i++)
        {
            s=rag(i);
            if(prime(i) && prime(s))
            System.out.print(i+",");
        }
    }
}
