public class Main
{
    public static int rag(int a,int b)
    {
        int big=Math.max(a,b);
        int small=Math.min(a,b);
        int s=big;
        while(true)
        {
            if(big%small==0)
            break;
            big=big+s;
        }
        return big;
    }
	public static void main(String[] args) 
	{
        int []a={8,10,4,15,12,18,20};
        int lcm=a[0];
        for(int i=1;i<a.length;i++)
        
        {
            lcm=rag(lcm,a[i]);
        }
        System.out.println("THE LCM IS "+lcm);
	}
}
